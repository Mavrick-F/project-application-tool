# ============================================
# DATASET CONFIGURATION FOR MEMPHIS MPO PROJECT APPLICATION TOOL
# ============================================
# This file defines all datasets used in spatial analysis.
# To add a new dataset: copy an example below, paste it, and change the values.
# No other files need to be edited — just save and refresh the browser.
#
# IMPORTANT: Indentation matters in YAML. Use spaces, not tabs.
# Each level of indentation must be consistent (2 spaces recommended).
#
# ============================================
# GEOMETRY TYPES
# ============================================
#   Point      - Single coordinate locations (crashes, hospitals, bus stops)
#   LineString - Road segments, routes, corridors
#   Polygon    - Areas (zones, districts, wetlands)
#
# ============================================
# ANALYSIS METHODS AND REQUIRED FIELDS
# ============================================
# Pick the method that matches what you want to know, then provide the fields it needs.
#
#  METHOD                    | WHAT IT DOES                                              | FIELDS REQUIRED
#  --------------------------|-----------------------------------------------------------|-------------------------------------------------
#  listParallelFeatures      | Lines running alongside the project (roads, routes)       | bufferDistance, minSharedLength
#  averageParallelValue      | Length-weighted average of a number from parallel lines   | bufferDistance, minSharedLength, averageField
#  listIntersectingFeatures  | Areas/lines that overlap the project geometry             | (none — uses direct geometry intersection)
#  listNearbyFeatures        | Points or polygons within a set distance                  | proximityBuffer
#  countByCategory           | Count nearby points, grouped by a field (e.g. severity)  | proximityBuffer, countByField (optional)
#  hasNearbyFeatures         | Yes/No — does anything exist within the buffer?           | proximityBuffer
#  measureProjectByCategory  | Total project length broken down by a status field        | bufferDistance, minSharedLength, statusField
#  measureIntersectedArea    | Area of polygon features that overlap the project         | bufferDistance or proximityBuffer
#  projectCoverage           | % of project that runs along a reference network          | bufferDistance
#  sumNearbyValues           | Sum a numeric field from features within buffer            | proximityBuffer, sumField
#  findNearestFeatures       | Distance-ranked list of nearest features                  | nearestCount, maxDistance (optional)
#  measureNearbyArea         | Sum an area field from nearby polygons (acres, sq mi, etc) | proximityBuffer, sumField, sumUnit (optional)
#
# ALL METHODS also require under properties:
#   displayField  - The field name from your data file shown in results (case-sensitive)
#
# ============================================
# RESULT STYLES  (controls how results are displayed in the panel and PDF)
# ============================================
#   list          - Bulleted list of matched feature names          (use with: listParallelFeatures, listIntersectingFeatures, listNearbyFeatures)
#   count         - Total count with breakdown by category          (use with: countByCategory)
#   binary        - Yes / No                                        (use with: hasNearbyFeatures)
#   lengthByStatus - Length of project per status category           (use with: measureProjectByCategory)
#   area          - Total area with list of features                (use with: measureIntersectedArea)
#   percentage    - Percent of project covered                      (use with: projectCoverage)
#   sum           - Numeric total (add sumUnit: 'acres' to label units) (use with: sumNearbyValues, measureNearbyArea)
#   nearest       - Ranked list with distances                      (use with: findNearestFeatures)
#   averageValue  - Single averaged value with color-coded tier     (use with: averageParallelValue)
#   table         - Multi-column table (requires additionalFields)  (use with: any method)
#
# ============================================
# STYLE OPTIONS
# ============================================
# color / fillColor: Hex color code ('#FF0000') or named color ('red')
# opacity / fillOpacity: 0.0 (invisible) to 1.0 (solid). Typical: 0.3–0.9
# weight: Line thickness in pixels (1–5 typical). Points use radius instead.
#
# ============================================
# OPTIONAL PROPERTIES (add only if needed)
# ============================================
# additionalFields: []         - Extra field names to show alongside displayField
# staticLabel: 'Some Text'     - Override field value with a fixed label for all features
# showFullLayerInPdf: true     - Show entire layer on PDF map, not just matched features
# lazyLoad: true               - Only load data when analysis runs (use for large datasets)
# featureServiceUrl: 'https://'- ArcGIS Feature Service URL (alternative to filePath)
# styleByProperty:             - Color features differently based on a field value (see LOTTR example)
# displayThresholds:           - Color-coded bands for averageValue results (see LOTTR example)
# analysisFilter:              - Only analyze features matching a field/value condition
# offsetByDirection:             - Offset features so directional pairs don't overlap on map
#   field: direction             - Property field containing direction value
#   positiveValues: [...]        - Direction values that get positive pixel offset (others get negative)
#   offset: 2                    - Offset amount in pixels (default: 2)
# maxRecords: 5000             - Maximum features to request from Feature Service (default: 5000)
# fieldLabels:                 - Rename raw field names for display (e.g. F__Below_A: '% Below ALICE')
# areaLabel: 'Wetlands'        - Description appended to area totals (e.g. 'Total: 4.2 acres of Wetlands')
# bufferUnit: 'feet'           - Unit for bufferDistance, proximityBuffer, minSharedLength, maxDistance values.
#                                Supported: 'feet' (default), 'miles', 'meters', 'kilometers'
#                                Example: bufferUnit: miles with bufferDistance: 0.5 means a half-mile buffer.
# resultUnit:                  - Unit for output measurements displayed in results panel and PDF.
#                                For distance methods (findNearestFeatures): 'feet' (default), 'miles', 'meters', 'kilometers'
#                                For area methods (measureIntersectedArea, measureNearbyArea): 'acres' (default), 'sq miles', 'sq ft', 'sq meters', 'sq km'
#                                For length methods (measureProjectByCategory): 'miles' (default), 'feet', 'kilometers', 'meters'
# sumUnit: 'acres'             - Unit label appended to sum results for sumNearbyValues and measureNearbyArea.
#                                Use any label string (e.g. 'acres', 'sq miles', 'jobs', 'residents').
# specialHandling:
#   removeDirectionalSuffixes: true  - Remove " NB"/" SB"/" EB"/" WB"/" IB"/" OB" from route names
#   deduplicate: true                - Merge features with the same display name
#
# ============================================
